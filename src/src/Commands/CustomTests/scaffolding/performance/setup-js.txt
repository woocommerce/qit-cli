import { check } from "k6";
import http from "k6/http";

// Setup function runs once before the test starts
export function setup() {
    const baseUrl = __ENV.BASE_URL || "http://localhost";
    
    console.log("Setting up performance test environment...");
    console.log(`Target URL: ${baseUrl}`);
    
    // Verify the site is accessible before starting the test
    const response = http.get(baseUrl);
    
    if (!check(response, {
        "site is accessible during setup": (r) => r.status === 200,
    })) {
        throw new Error(`Site at ${baseUrl} is not accessible. Aborting test.`);
    }
    
    // Return data that will be passed to the test functions
    return {
        baseUrl: baseUrl,
        setupTime: new Date().toISOString(),
        siteInfo: {
            status: response.status,
            responseTime: response.timings.duration,
        },
    };
}
