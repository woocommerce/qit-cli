import { check, sleep } from "k6";
import http from "k6/http";
import { qitHelpers } from "/qitHelpers/qitHelpers.js";

export let options = {
    stages: [
        { duration: "10s", target: 5 },   // Ramp up to 5 users over 10 seconds
        { duration: "30s", target: 10 },  // Stay at 10 users for 30 seconds
        { duration: "10s", target: 0 },   // Ramp down to 0 users over 10 seconds
    ],
    thresholds: {
        // 95% of requests should be below 500ms
        http_req_duration: ["p(95)<500"],
        // Error rate should be below 1%
        http_req_failed: ["rate<0.01"],
        // 99% of requests should be below 1000ms
        "http_req_duration{name:homepage}": ["p(99)<1000"],
        // Shop page should be fast
        "http_req_duration{name:shop}": ["p(95)<800"],
    },
};

export default function() {
    const baseUrl = __ENV.BASE_URL || "http://localhost";
    
    // Test homepage performance
    let response = http.get(baseUrl, {
        tags: { name: "homepage" },
    });
    
    check(response, {
        "homepage status is 200": (r) => r.status === 200,
        "homepage loads in < 500ms": (r) => r.timings.duration < 500,
    });
    
    qitHelpers.utils.waitFor(1);
    
    // Test WooCommerce shop page
    response = http.get(`${baseUrl}/shop/`, {
        tags: { name: "shop" },
    });
    
    check(response, {
        "shop page status is 200": (r) => r.status === 200,
        "shop page loads in < 800ms": (r) => r.timings.duration < 800,
        "shop page contains products": (r) => r.body.includes("woocommerce"),
    });
    
    qitHelpers.utils.waitFor(1);
    
    // Test cart functionality
    response = http.get(`${baseUrl}/cart/`, {
        tags: { name: "cart" },
    });
    
    check(response, {
        "cart page status is 200": (r) => r.status === 200,
        "cart page loads in < 600ms": (r) => r.timings.duration < 600,
    });
    
    qitHelpers.utils.waitFor(1);
    
    // Test checkout page
    response = http.get(`${baseUrl}/checkout/`, {
        tags: { name: "checkout" },
    });
    
    check(response, {
        "checkout page status is 200": (r) => r.status === 200,
        "checkout page loads in < 700ms": (r) => r.timings.duration < 700,
    });
    
    // Random sleep between 1-3 seconds to simulate real user behavior
    sleep(Math.random() * 2 + 1);
}
