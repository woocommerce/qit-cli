import { check, sleep } from "k6";
import http from "k6/http";
import { qitHelpers } from "/qitHelpers/qitHelpers.js";

export let options = {
    scenarios: {
        spike_test: {
            executor: "ramping-vus",
            stages: [
                { duration: "10s", target: 5 },     // Normal load
                { duration: "10s", target: 100 },   // Spike to 100 users
                { duration: "30s", target: 100 },   // Stay at spike
                { duration: "10s", target: 5 },     // Return to normal
                { duration: "10s", target: 0 },     // Ramp down
            ],
        },
    },
    thresholds: {
        // More lenient thresholds for spike testing
        http_req_duration: ["p(95)<2000"],
        http_req_failed: ["rate<0.1"],
        // Ensure we can handle the spike
        "http_req_duration{scenario:spike_test}": ["p(99)<5000"],
    },
};

export default function() {
    const baseUrl = __ENV.BASE_URL || "http://localhost";
    
    // Simple but effective spike test focusing on critical pages
    
    // Test homepage under load
    let response = http.get(baseUrl, {
        tags: { name: "homepage", scenario: "spike_test" },
    });
    
    check(response, {
        "homepage survives spike": (r) => r.status === 200,
        "homepage response time acceptable": (r) => r.timings.duration < 3000,
    });
    
    // Brief pause
    qitHelpers.utils.waitFor(0.5);
    
    // Test shop page under load
    response = http.get(`${baseUrl}/shop/`, {
        tags: { name: "shop", scenario: "spike_test" },
    });
    
    check(response, {
        "shop page survives spike": (r) => r.status === 200,
        "shop page response time acceptable": (r) => r.timings.duration < 4000,
    });
    
    // Test cart endpoint
    response = http.get(`${baseUrl}/cart/`, {
        tags: { name: "cart", scenario: "spike_test" },
    });
    
    check(response, {
        "cart page survives spike": (r) => r.status === 200,
    });
    
    // Minimal sleep to maximize concurrent requests during spike
    qitHelpers.utils.waitFor(0.1);
}
