import { check } from "k6";
import http from "k6/http";

// Teardown function runs once after the test completes
export function teardown(data) {
    console.log("Tearing down performance test environment...");
    console.log(`Test started at: ${data.setupTime}`);
    console.log(`Test completed at: ${new Date().toISOString()}`);
    
    // Optional: Verify the site is still accessible after the test
    if (data.baseUrl) {
        const response = http.get(data.baseUrl);
        
        check(response, {
            "site still accessible after test": (r) => r.status === 200,
        });
        
        if (response.status === 200) {
            console.log("✓ Site is still accessible after performance test");
        } else {
            console.log("⚠ Site accessibility check failed after test");
        }
    }
    
    console.log("Performance test teardown complete.");
}
